# API Design

This document defines the first public API shape for Endnotes.

The goal is to keep the API small enough to remember while still supporting real production use cases.

## Exports

```tsx
import {
  EndnotesProvider,
  Endnote,
  Endnotes,
  useEndnotes,
} from "endnotes"
```

## `EndnotesProvider`

Wraps a page, article, generated response, or app section where references are used.

```tsx
<EndnotesProvider adapt>
  {children}
</EndnotesProvider>
```

### Props

```ts
type EndnotesProviderProps = {
  children: React.ReactNode
  adapt?: boolean
  theme?: EndnotesTheme
  interaction?: "scroll" | "popover" | "drawer" | "sidenote"
  marker?: "superscript" | "bracket" | "pill"
  strict?: boolean
  scope?: string
  dedupe?: "id" | "href" | "none"
}
```

### Defaults

```ts
const defaults = {
  adapt: false,
  interaction: "popover",
  marker: "superscript",
  strict: false,
  dedupe: "id",
}
```

## `Endnote`

Renders an inline reference marker and registers a source with the nearest provider.

```tsx
<Endnote
  href="https://example.com/report"
  title="Example Report"
  source="Example Research"
/>
```

### Props

```ts
type EndnoteProps = {
  id?: string
  href?: string
  title: string
  source?: string
  author?: string
  date?: string
  accessed?: string
  quote?: string
  supports?: string
  description?: string
  type?: "article" | "paper" | "book" | "report" | "website" | "dataset" | "internal" | "other"
  marker?: "superscript" | "bracket" | "pill"
  className?: string
}
```

### Identity

Endnotes should resolve identity in this order:

1. `id`
2. `href`
3. generated instance ID

If an `id` is provided, it is the canonical identity for that source.

## `Endnotes`

Renders the generated list of registered notes.

```tsx
<Endnotes />
```

### Props

```ts
type EndnotesProps = {
  title?: string
  variant?: "list" | "cards" | "compact"
  showBacklinks?: boolean
  showCopy?: boolean
  className?: string
}
```

### Defaults

```ts
const defaults = {
  title: "Endnotes",
  variant: "list",
  showBacklinks: true,
  showCopy: true,
}
```

## `useEndnotes`

Allows custom renderers and generated content systems to interact with the registry.

```tsx
const {
  notes,
  cite,
  getNoteById,
  scrollToNote,
  scrollToReference,
} = useEndnotes()
```

### Shape

```ts
type UseEndnotesReturn = {
  notes: RegisteredEndnote[]
  cite: (source: EndnoteSource) => RegisteredEndnote
  getNoteById: (id: string) => RegisteredEndnote | undefined
  scrollToNote: (id: string) => void
  scrollToReference: (id: string, referenceId?: string) => void
}
```

## Source model

```ts
type EndnoteSource = {
  id?: string
  href?: string
  title: string
  source?: string
  author?: string
  date?: string
  accessed?: string
  quote?: string
  supports?: string
  description?: string
  type?: "article" | "paper" | "book" | "report" | "website" | "dataset" | "internal" | "other"
}
```

## Registered model

```ts
type RegisteredEndnote = EndnoteSource & {
  id: string
  number: number
  references: InlineReference[]
}

type InlineReference = {
  id: string
  noteId: string
  order: number
  element?: HTMLElement | null
}
```

## Theme model

```ts
type EndnotesTheme = {
  background?: string
  foreground?: string
  muted?: string
  border?: string
  accent?: string
  popover?: string
  popoverForeground?: string
  radius?: string
  fontFamily?: string
  shadow?: string
}
```

## Example: AI answer

```tsx
<EndnotesProvider adapt interaction="popover" marker="pill">
  <p>
    Claim text generated by an AI system.
    <Endnote
      id="source-1"
      href="https://example.com"
      title="Example Source"
      source="Example"
      supports="The specific claim in this sentence."
    />
  </p>

  <Endnotes title="Sources" variant="cards" />
</EndnotesProvider>
```

## Example: editorial page

```tsx
<EndnotesProvider adapt interaction="scroll" marker="superscript">
  <article>
    <p>
      A longer essay sentence with a reference.
      <Endnote href="https://example.com" title="Example Source" />
    </p>

    <Endnotes title="Notes" variant="compact" />
  </article>
</EndnotesProvider>
```
